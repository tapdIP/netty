    @BeforeAll
    @DisplayName("Setup test environment")
    void setupAll(Vertx vertx, VertxTestContext testContext) throws IOException {
        this.vertx = vertx;
        
        // Create temporary directory and test files
        tempDir = Files.createTempDirectory("static-test");
        createTestFiles();
        
        // Create WebClient (don't follow redirects automatically)
        client = WebClient.create(vertx, new WebClientOptions()
                .setDefaultHost(TEST_HOST)
                .setDefaultPort(TEST_PORT)
                .setFollowRedirects(false));
        
        // Setup and start server
        Router router = setupRouter(vertx);
        server = vertx.createHttpServer()
                .requestHandler(router)
                .listen(TEST_PORT)
                .onComplete(testContext.succeedingThenComplete());
    }

    @AfterAll
    @DisplayName("Cleanup test environment")
    void tearDownAll(VertxTestContext testContext) throws IOException {
        // Clean up test files
        deleteDirectory(tempDir.toFile());
        
        // Close server and client
        if (server != null) {
            server.close()
                    .compose(v -> client != null ? client.close() : vertx.closeFuture())
                    .onComplete(testContext.succeedingThenComplete());
        } else {
            testContext.completeNow();
        }
    }

    /**
     * Setup router with static resource configuration
     */
    private Router setupRouter(Vertx vertx) {
        Router rootRouter = Router.router(vertx);
        FeatureContext ctx = new FeatureContext(vertx, rootRouter);
        
        configureStaticResources(ctx, PREFIX, INDEX, tempDir.toString(), 
                false, false, true, true, 3600);
        
        return rootRouter;
    }

    /**
     * Configure static resources (your original code)
     */
    private void configureStaticResources(FeatureContext ctx, String prefix, 
            String index, String root, boolean includeHidden, boolean directoryListing,
            boolean cachingEnabled, boolean longCache, int maxAgeSeconds) {
        
        // Add root path redirect handler
        ctx.rootRouter().route("/").handler(rc -> {
            rc.response()
                .putHeader("Location", prefix + "/")
                .setStatusCode(302)
                .end();
        });

        final Router staticRouter = Router.router(ctx.vertx());

        // Fix 1: Fix path check logic to prevent path traversal attacks
        staticRouter.route().handler(rc -> {
            final String p = rc.normalizedPath();
            // Fix: Check dangerous characters instead of checking "/"
            if (p != null && (p.contains("..") || p.contains("\\") || p.contains("//"))) {
                rc.response().setStatusCode(404).end();
                return;
            }
            rc.next();
        });

        staticRouter.route().handler(rc -> {
            final String p = rc.normalizedPath();
            rc.addHeadersEndHandler(v -> {
                if (!cachingEnabled) return;
                if (equalsPath(p, "/") || equalsPath(p, "/" + index)) {
                    // Fix 2: Fix rc.response to rc.response()
                    rc.response().putHeader("Cache-Control", "no-cache");
                } else if (longCache && isFingerprinted(p)) {
                    // Fix 2: Fix rc.response to rc.response()
                    rc.response().putHeader("Cache-Control", "public, max-age=31536000, immutable");
                } else {
                    // Fix 2: Fix rc.response to rc.response()
                    rc.response().putHeader("Cache-Control", "public, max-age=" + maxAgeSeconds);
                }
            });
            rc.next();
        });

        final StaticHandler sh = StaticHandler.create(root)
                .setIncludeHidden(includeHidden)
                .setDirectoryListing(directoryListing)
                .setCachingEnabled(cachingEnabled)
                .setMaxAgeSeconds(maxAgeSeconds)
                .setDefaultContentEncoding("UTF-8")
                .setIndexPage(index);

        staticRouter.route("/*").handler(sh);

        ctx.rootRouter().route(prefix.endsWith("/*") ? prefix : (prefix + "/*"))
                .subRouter(staticRouter);
    }

    /**
     * Check if path equals another path
     */
    private boolean equalsPath(String path1, String path2) {
        return path1 != null && path1.equals(path2);
    }

    /**
     * Check if file is fingerprinted
     */
    private boolean isFingerprinted(String path) {
        if (path == null) return false;
        return path.matches(".*\\.[a-f0-9]{8,}\\.(js|css|png|jpg|jpeg|gif|svg|woff|woff2|ttf|eot)$");
    }

    /**
     * Delete directory recursively
     */
    private void deleteDirectory(File directory) throws IOException {
        if (directory.exists()) {
            File[] files = directory.listFiles();
            if (files != null) {
                for (File file : files) {
                    if (file.isDirectory()) {
                        deleteDirectory(file);
                    } else {
                        Files.delete(file.toPath());
                    }
                }
            }
            Files.delete(directory.toPath());
        }
    }

    /**
     * Mock FeatureContext
     */
    static class FeatureContext {
        private final Vertx vertx;
        private final Router rootRouter;

        public FeatureContext(Vertx vertx, Router rootRouter) {
            this.vertx = vertx;
            this.rootRouter = rootRouter;
        }

        public Vertx vertx() {
            return vertx;
        }

        public Router rootRouter() {
            return rootRouter;
        }
    }
